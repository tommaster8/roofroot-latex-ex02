\chapter{בקשת ההשקעה וסיכום}\label{ch:investment}

\section{הבקשה}
גגון מגייסת \RRseed{} מיליון \shekel{} בסבב \en{Seed} ב-2027. הסכום נועד להוכיח את הכלכלה של הגג בשטח, לבנות את \en{RootOS} לרמת ייצור, ולהגיע ל-\RRroofsFirst{} גגות פעילים בשנה הראשונה ולבסיס להמשך. סבב \en{Series A} של \RRseriesA{} מיליון \shekel{} מתוכנן ל-2029; יחד, סך ההון הוא \RRequityTotal{} מיליון \shekel.

\section{אבני דרך עד סבב \en{Series A}}
\begin{enumerate}
  \item \textbf{2027:} התקנת \RRroofsFirst{} גגות ראשונים, חתימת הסכמי ועד, פיילוט בגוש דן, והשקת גרסת הייצור של \en{RootOS}.
  \item \textbf{2028:} הוכחת היחידה הכלכלית בשטח: תפוקה, מחיר ופחת מול הנחות המודל (טבלה~\ref{tab:pnl}), והפעלת מנוי ״סל ירוק״ ראשון.
  \item \textbf{2029:} סבב \en{Series A}, מכירת רישיון \en{RootOS} לחוות חיצוניות, והאצת קצב ההתקנה.
  \item \textbf{\RRbreakEvenYear:} \en{EBITDA} חיובי.
  \item \textbf{2031:} \RRroofsFinal{} גגות והכנסות של \RRrevFinal{} מיליון \shekel.
\end{enumerate}

\begin{riskbox}[מה יכול לעכב]
עיכוב בהסכמי ועד, או תפוקה נמוכה מהצפוי, יאריכו את תקופת ההחזר. הרזרבה וניתוח הרגישות (פרק~\ref{ch:financials}) נועדו לתת לנו מרווח תמרון, והתוכנית כולה תלויה בהוכחת הנחות היסוד בפיילוט.
\end{riskbox}

\section{חזון}
גג אחד הוא ירקות לשכונה. \RRroofsFinal{} גגות הם רשת מזון מקומית בגוש דן, עם \en{RootOS} כשכבת ניהול שאפשר לרשת לכל חווה עירונית. אנו מזמינים אתכם להצטרף לסבב ה-\en{Seed}, ולהפוך את הגגות של העיר למקור המזון הטרי שלה.

\begin{insightbox}[סיכום]
\RRseed{} מיליון \shekel{} עכשיו מוכיחים כלכלה של \RRroofPayback{} שנים לגג; \RRseriesA{} מיליון \shekel{} ב-2029 מגדילים את התוכנית ל-\RRroofsFinal{} גגות ולהכנסות של \RRrevFinal{} מיליון \shekel{} ב-2031.
\end{insightbox}
